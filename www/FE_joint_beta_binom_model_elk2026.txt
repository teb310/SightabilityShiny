model
{
# Priors
	## Detection model
    	bo ~ dnorm(0, 0.1) 		# vaguely-defined parameters
    	bvoc ~ dnorm(0, 0.1) 	# vaguely-defined parameters
    
	## Detection/non-detection data
    	for (i in 1:R) { 	# for each detection/non-detection of collared individual
        	logitg[i] <- bo + bvoc * x.tilde[i] 	# logistic regression equal to intercept + 									coefficient*estimated voc
        	g.tilde[i] <- exp(logitg[i])/(1 + exp(logitg[i]))
        	z.tilde[i] ~ dbern(g.tilde[i])
    }
    	## Visual obstruction
    	for (j in 1:Nyears) {
        	for (i in 1:Nstrata) {		# for each class(stratum)-year

		### Hyperpriors for alpha.voc and beta.voc
            	mu.voc[i,j] ~ dunif(0.01, 0.99) 	# hyperprior 1 is uniform between 0 and 1
            	lnrho.voc[i,j] ~ dnorm(5,1)   T(0.01,10) 	# nat log of hyperprior 2 is normal w/ mean 5 and 								sd 1 (truncated at 0 and 10)
            	rho.voc[i,j] <- exp(lnrho.voc[i,j]) 	# hyperprior 2

		### alpha and beta for later beta distribution of voc
            	alpha.voc[i,j] <- mu.voc[i,j]*rho.voc[i,j]
            	beta.voc[i,j] <- rho.voc[i,j]-(mu.voc[i,j]*rho.voc[i,j])
        }
    }
    
# Priors for psi
  for(k in 1:Nyears){
    for(i in 1:Nstrata){ 	# for each stratum-year
	## Hyperpriors for alpha psi and beta psi
      	mu.psi[i,k] ~ dunif(0.01,0.99)
      	lnrho.psi[i,k] ~ dnorm(5,1)	T(0.01,10)
      	rho.psi[i,k] <- exp(lnrho.psi[i,k])

	## Alpha and beta used in beta distribution of psi later on
      	alpha.psi[i,k] <- mu.psi[i,k]*rho.psi[i,k]
      	beta.psi[i,k] <- rho.psi[i,k]-(mu.psi[i,k]*rho.psi[i,k])
    }
  }

# The probability that each group "belongs to N" (data augmentation)
  for(i in 1:Nstrata.subunit.yr){
    	psi[i] ~ dbeta(alpha.psi[h.plots[i],yr.plots[i]], 					beta.psi[h.plots[i],yr.plots[i]])	T(0.0001,0.99)
  	}


# Priors for groupsize - make dependent on year and stratum (but sample all subunits)
for (m in 1:Nyears) {
        for (p in 1:Nstrata) {
            	loglam[p,m] ~ dnorm(0, 0.1)  T(-10, 10)
            	lam[p,m] <- exp(loglam[p, m])
            	r[p,m] ~ dgamma(1,0.01)
        	}
	}

# Final likelihoods
    for (t in 1:Ngroups) {

	x[t] ~ dbeta(alpha.voc[h[t],yr[t]],beta.voc[h[t],yr[t]])	T(0,0.999) 	# voc is beta distribution

        q[t] ~ dbin(psi[plot[t]], 1) 	# whether or not group belongs to population is binomial based on psi

        g[t] <- exp(bo + bvoc * x[t])/(1 + exp(bo + bvoc * x[t])) 	# probability of seeing group given voc

        pseen[t] <- q[t] * g[t] 	# probability of seeing group is 0 if group doesn't belong to population

        z[t] ~ dbin(pseen[t], 1) 	# final determination of whether augmented group is seen

        ## Negative binomial distribution for group size (minus 1)
	p.nb[t] <- r[h[t], yr[t]] / (r[h[t], yr[t]] + lam[h[t], yr[t]]) 	# defining shape for dnbinom

        ym1[t] ~ dnbinom(p.nb[t], r[h[t], yr[t]]) 	# group size is negative binomial

        y[t] <- ym1[t] + 1	# add 1 to get final group size 

	## Group size is 1*y if group is part of population and zero otherwise
        taus[t] <- q[t] * y[t]
    	}

# Defining ranges for calculating abundance per subunit-stratum-year
for (u in 1:Nstrata.subunit.yr) { 	# for each subunit-stratum-year
	t[u] <- scalars[u,1] 	# 'from' value is in first column
	d[u] <- scalars[u,2] 	# 'to' value in second column
	tau.hat[u] <- sum(taus[c(t[u]:d[u])])	# sum up group sizes per subunit-stratum-year
	}

# Defining ranges for calculating abundance per subunit-year (ignoring strata)
for (v in 1:Nsubunit.yr) {	# for each subunit-year
	u[v] <- total.scalars[v,1]	# 'from' value is in first column
	w[v] <- total.scalars[v,2]	# 'to' value in second column
	total.tau.hat[v] <- sum(taus[c(u[v]:w[v])])	# sum up group sizes per subunit-year

	}
}
